import ollama
import json
import time

def generate_health_insights(input_data):
    """Improved version with better prompt engineering and parsing"""
    # Structured prompt with explicit formatting instructions
    prompt = f"""
    **Comprehensive Health Risk Assessment & Clinical Decision Support**

    Synthesize the patient's clinical data into a structured JSON response using current evidence-based guidelines. Prioritize precision, numerical specificity, and non-repetitive actionable directives. Maintain strict adherence to the output format.

    **Patient Dataset:**
    {json.dumps(input_data, indent=2)}

    **Required JSON Structure:**
    ```json
    {{
        "recommendations": {{
            "lifestyle": ["Implement <specific intervention> with <measurable parameter>", ...],
            "monitoring": ["Perform <test> every <frequency> using <method>", ...],
            "medication": ["Consider <drug class> for <indication> (<dose range>)", ...]
        }},
        "insights": {{
            "primary_risk_factors": ["<metric>: <value> (<severity classification>)", ...],
            "trend_analysis": ["<biomarker> changed by <delta> over <timeframe>", ...],
            "historical_comparisons": ["<parameter> vs <prior_value> (<time_interval>)", ...],
            "comorbidities_interactions": ["<condition> influencing <risk_factor>"]
        }},
        "action_items": {{
            "immediate_actions": ["Urgent <action> within <timeframe>", ...],
            "short_term_goals": ["Achieve <metric> <target> by <duration>", ...],
            "long_term_goals": ["Sustain <metric> <target_range> through <strategy>", ...],
            "follow_up_schedule": ["<Specialist> evaluation every <interval>", ...]
        }},
        "summary": "Concise synthesis (<75 words) emphasizing critical risks and intervention priorities"
    }}

    **Composition Guidelines:**
    
    < > tags can be removed when placed with actual values from the data, also sample examples provided in values of subkeys should not be given as output only take it as reference for the output format.

    1. **Terminology & Precision:**
    - Use active verbs: Implement, Achieve, Monitor, Consider
    - Include numerical thresholds (e.g., BP >140/90 mmHg)
    - Reference clinical standards (JNC, ADA, ACC/AHA)

    2. **Temporal Specificity:**
    - Define clear timeframes: "Within 72h", "Every 6-8 weeks"
    - Quantify trends: "12% reduction over 6 months"

    4. **Intervention Design:**
    - Pair actions with metrics: "Reduce sodium intake to <2.4g/day"
    - Specify monitoring modalities: "Home BP log 2x daily"

    5. **Differentiation:**
    - Avoid overlapping recommendations across categories
    - Ensure medication suggestions match risk severity

    **Validation Requirements:**
    - Cross-validate biomarker trends with historical data
    - Align medication suggestions with current BMI/labs
    - Confirm temporal consistency in follow-up scheduling

    Generate a response that enables immediate clinical action while establishing measurable health targets.
    """


    try:
        response = ollama.chat(
            model='llama3.2:1b',
            messages=[{'role': 'user', 'content': prompt}],
            format='json',
            options={
                'temperature': 0.1,
                'num_ctx': 4096,
                'max_tokens': 5000,
                'stop': ["\n```"],
            }
        )
        
        # Extract JSON from markdown code block
        raw_response = response['message']['content']
        return json.loads(raw_response)
    
    except (KeyError, IndexError, json.JSONDecodeError) as e:
        print(f"Parsing error: {str(e)}")
        return {"error": "Failed to process model response"}
    except Exception as e:
        print(f"API error: {str(e)}")
        return {"error": "Model inference failed"}
